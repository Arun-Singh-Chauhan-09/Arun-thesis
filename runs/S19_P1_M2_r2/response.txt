```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: ssd-batch-deployment
spec:
  selector:
    matchLabels:
      app: ssd-batch
  replicas: 1
  template:
    metadata:
      labels:
        app: ssd-batch
    spec:
      nodeSelector:
        disktype: ssd
      tolerations:
      - key: dedicated
        operator: Equal
        value: batch
        effect: NoSchedule
      containers:
      - name: app
        image: nginx:1.14.2
        ports:
        - containerPort: 80
```
