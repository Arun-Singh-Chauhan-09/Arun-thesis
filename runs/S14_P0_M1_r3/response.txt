```yaml
apiVersion: v1
kind: Pod
metadata:
  name: logging-pod
spec:
  containers:
  - name: main-container
    image: busybox
    volumeMounts:
    - name: logs
      mountPath: /logs
    command: ["sh", "-c"]
    args:
    - while true; do echo "$(date)" >> /logs/app.log; sleep 1; done
  - name: sidecar-container
    image: busybox
    volumeMounts:
    - name: logs
      mountPath: /logs
    command: ["sh", "-c"]
    args:
    - tail -f /logs/app.log
  volumes:
  - name: logs
    emptyDir: {}
---
apiVersion: v1
kind: PersistentVolume
metadata:
  name: logging-pv
spec:
  capacity:
    storage: 1Gi
  accessModes:
    - ReadOnlyMany
  persistentVolumeReclaimPolicy: Retain
  local:
    path: /mnt/data
  storageClassName: local-storage
  nodeAffinity:
    required:
      nodeSelectorTerms:
      - matchExpressions:
        - key: kubernetes.io/hostname
          operator: In
          values:
          - your-node-name
---
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: logging-pvc
spec:
  accessModes:
    - ReadOnlyMany
  resources:
    requests:
      storage: 1Gi
```
