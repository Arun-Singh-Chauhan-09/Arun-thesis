```yaml
---
apiVersion: v1
kind: Service
metadata:
  name: database
spec:
  clusterIP: None
  selector:
    app: database
---
apiVersion: v1
kind: Pod
metadata:
  name: app-with-init
  labels:
    app: app
spec:
  initContainers:
  - name: wait-for-db
    image: busybox:1.36
    command:
    - /bin/sh
    - -c
    - |
      echo "Waiting for database service to be resolvable..."
      until nslookup database; do
        echo "Database service not ready, retrying..."
        sleep 2
      done
      echo "Database service is resolvable."
  containers:
  - name: app
    image: nginx:1.25
    ports:
    - containerPort: 80
---
apiVersion: v1
kind: Pod
metadata:
  name: database
  labels:
    app: database
spec:
  containers:
  - name: db
    image: postgres:15
    env:
    - name: POSTGRES_DB
      value: mydb
    - name: POSTGRES_USER
      value: user
    - name: POSTGRES_PASSWORD
      value: password
```
