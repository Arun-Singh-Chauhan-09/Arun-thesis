```yaml
apiVersion: v1
kind: ConfigMap
metadata:
  name: app-config
data:
  config.json: |
    {
      "key": "value"
    }
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: app-deployment
spec:
  selector:
    matchLabels:
      app: app
  replicas: 1
  template:
    metadata:
      labels:
        app: app
    spec:
      containers:
      - name: app
        image: busybox
        volumeMounts:
        - name: config
          mountPath: /etc/config
        command: ["cat", "/etc/config/config.json"]
      volumes:
      - name: config
        configMap:
          name: app-config
```
