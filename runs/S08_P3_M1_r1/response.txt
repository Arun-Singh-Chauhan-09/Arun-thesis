```yaml
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: example-ingress
spec:
  ingressClassName: nginx
  rules:
  - host: app.example.com
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: hardened-app
            port:
              number: 8080
---
apiVersion: v1
kind: Service
metadata:
  name: hardened-app
spec:
  selector:
    app: hardened-app
  ports:
  - name: http
    port: 8080
    targetPort: 8080
  type: ClusterIP
```
