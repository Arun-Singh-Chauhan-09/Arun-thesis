apiVersion: batch/v1
kind: Job
metadata:
  name: db-migration-task
spec:
  backoffLimit: 2
  template:
    metadata:
      labels:
        app: db-migration-task
    spec:
      restartPolicy: OnFailure
      automountServiceAccountToken: false
      securityContext:
        runAsNonRoot: true
        runAsUser: 10001
        runAsGroup: 10001
        fsGroup: 10001
        seccompProfile:
          type: RuntimeDefault
      containers:
      - name: migration
        image: postgres:14
        imagePullPolicy: Always
        command: ["sh", "-c", "psql -h localhost -U postgres -d mydb -f /migration/sql/migrate.sql"]
        securityContext:
          runAsNonRoot: true
          allowPrivilegeEscalation: false
          readOnlyRootFilesystem: true
          capabilities:
            drop: ["ALL"]
          seccompProfile:
            type: RuntimeDefault
        resources:
          requests:
            cpu: "50m"
            memory: "64Mi"
          limits:
            cpu: "200m"
            memory: "128Mi"
        volumeMounts:
        - name: migration-scripts
          mountPath: /migration/sql
      volumes:
      - name: migration-scripts
        emptyDir: {}
---
apiVersion: v1
kind: ConfigMap
metadata:
  name: db-connection

  DB_HOST: "localhost"
  DB_USER: "postgres"
  DB_NAME: "mydb"
  DB_PASSWORD: "mysecretpassword"

  name: db-service

  name: db-pod

    app: db-pod

  - name: db

    - name: POSTGRES_USER
      value: "postgres"
    - name: POSTGRES_DB
      value: "mydb"

      value: "mysecretpassword"

      runAsNonRoot: true
