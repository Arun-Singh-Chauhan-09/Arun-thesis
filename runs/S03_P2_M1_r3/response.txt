```yaml
apiVersion: v1
kind: ConfigMap
metadata:
  name: app-config
data:
  CONFIG_KEY: config-value
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: config-app
spec:
  replicas: 1
  selector:
    matchLabels:
      app: config-app
  template:
    metadata:
      labels:
        app: config-app
    spec:
      containers:
      - name: app
        image: busybox
        command: ["sh", "-c"]
        args:
        - while true; do
            cat /config/config.txt;
            sleep 10;
          done
        volumeMounts:
        - name: config
          mountPath: /config
      volumes:
      - name: config
        configMap:
          name: app-config
          items:
          - key: CONFIG_KEY
            path: config.txt
```
