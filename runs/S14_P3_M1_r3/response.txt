```yaml
apiVersion: v1
kind: Pod
metadata:
  name: multi-container-pod
spec:
  containers:
  - name: main-container
    image: busybox
    command: ["sh", "-c"]
    args:
    - while true; do
        echo "$(date)" >> /shared/log.txt;
        sleep 1;
      done
    volumeMounts:
    - name: shared
      mountPath: /shared
  - name: sidecar-container
    image: busybox
    command: ["sh", "-c"]
    args:
    - tail -f /shared/log.txt
    volumeMounts:
    - name: shared
      mountPath: /shared
  volumes:
  - name: shared
    emptyDir: {}
---
```
