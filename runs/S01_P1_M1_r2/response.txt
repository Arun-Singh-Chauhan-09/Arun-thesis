```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: web-application
spec:
  selector:
    matchLabels:
      app: web-application
  replicas: 2
  template:
    metadata:
      labels:
        app: web-application
    spec:
      containers:
      - name: web-server
        image: nginx:latest
        ports:
        - containerPort: 80
---
apiVersion: v1
kind: Service
metadata:
  name: web-application-service
spec:
  selector:
    app: web-application
  ports:
  - name: http
    port: 80
    targetPort: 80
  type: ClusterIP
```
