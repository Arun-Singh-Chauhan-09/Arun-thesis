```yaml
apiVersion: v1
kind: Pod
metadata:
  name: app-with-log-sidecar
  labels:
    app: app-with-log-sidecar
spec:
  containers:
  - name: main-app
    image: busybox:1.36
    command:
    - /bin/sh
    - -c
    - >
      i=0;
      while true;
      do
        echo "$(date) INFO iteration=$i: main application log entry" >> /var/log/app/app.log;
        i=$((i+1));
        sleep 5;
      done
    volumeMounts:
    - name: log-volume
      mountPath: /var/log/app
  - name: log-sidecar
    image: busybox:1.36
    command:
    - /bin/sh
    - -c
    - tail -n+1 -F /var/log/app/app.log
    volumeMounts:
    - name: log-volume
      mountPath: /var/log/app
      readOnly: true
  volumes:
  - name: log-volume
    emptyDir: {}
```
