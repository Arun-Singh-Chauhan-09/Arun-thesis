```yaml
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: example-ingress
spec:
  ingressClassName: nginx
  rules:
  - host: app.example.com
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: web-application
            port:
              number: 80
---
apiVersion: v1
kind: Service
metadata:
  name: web-application
spec:
  selector:
    app: web-application
  ports:
  - name: http
    port: 80
    targetPort: 80
  type: ClusterIP
```
