```yaml
apiVersion: v1
kind: Pod
metadata:
  name: logging-pod
spec:
  containers:
  - name: main-container
    image: busybox
    command: ["sh", "-c"]
    args: ["while true; do echo 'INFO: logging information' >> /var/log/app.log; sleep 1; done"]
    volumeMounts:
    - name: logs
      mountPath: /var/log
  - name: sidecar-container
    image: busybox
    command: ["sh", "-c"]
    args: ["tail -f /var/log/app.log"]
    volumeMounts:
    - name: logs
      mountPath: /var/log
  volumes:
  - name: logs
    emptyDir: {}
---
```
