```yaml
apiVersion: v1
kind: Service
metadata:
  name: stable-identity
spec:
  clusterIP: None
  selector:
    app: stable-identity
  ports:
  - name: http
    port: 80
    targetPort: 8080

apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: stable-identity
spec:
  serviceName: stable-identity
  replicas: 3
  selector:
    matchLabels:
      app: stable-identity
  template:
    metadata:
      labels:
        app: stable-identity
    spec:
      containers:
      - name: app
        image: nginx:latest
        ports:
        - containerPort: 8080
        volumeMounts:
        - name: www
          mountPath: /usr/share/nginx/html
  volumeClaimTemplates:
  - metadata:
      name: www
    spec:
      accessModes: ["ReadWriteOnce"]
      resources:
        requests:
          storage: 1Gi
```
