```yaml
apiVersion: batch/v1
kind: CronJob
metadata:
  name: nightly-backup
spec:
  schedule: "0 2 * * *"
  successfulJobsHistoryLimit: 3
  failedJobsHistoryLimit: 1
  jobTemplate:
    spec:
      backoffLimit: 2
      template:
        metadata:
          labels:
            app: nightly-backup
        spec:
          restartPolicy: OnFailure
          containers:
          - name: backup
            image: busybox:1.36
            imagePullPolicy: Always
            command: ["sh", "-c", "echo 'backup started'; tar -czf /tmp/backup-$(date +%Y%m%d-%H%M%S).tar.gz /data 2>/dev/null || echo 'no data dir, simulating backup'; echo 'backup complete'"]
            resources:
              requests:
                cpu: "100m"
                memory: "128Mi"
              limits:
                cpu: "250m"
                memory: "256Mi"
            securityContext:
              runAsNonRoot: true
              allowPrivilegeEscalation: false
              readOnlyRootFilesystem: true
              capabilities:
                drop: ["ALL"]
              seccompProfile:
                type: RuntimeDefault
            volumeMounts:
            - name: tmp
              mountPath: /tmp
            - name: data
              mountPath: /data
              readOnly: true
          automountServiceAccountToken: false
          securityContext:
            runAsNonRoot: true
            runAsUser: 10001
            runAsGroup: 10001
            fsGroup: 10001
            seccompProfile:
              type: RuntimeDefault
          volumes:
          - name: tmp
            emptyDir: {}
          - name: data
            emptyDir: {}
```
