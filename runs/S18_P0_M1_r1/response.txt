```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: spread-application
spec:
  replicas: 3
  selector:
    matchLabels:
      app: spread-application
  template:
    metadata:
      labels:
        app: spread-application
    spec:
      topologySpreadConstraints:
      - maxSkew: 1
        topologyKey: kubernetes.io/hostname
        whenUnsatisfiable: DoNotSchedule
        labelSelector:
          matchLabels:
            app: spread-application
      containers:
      - name: application
        image: busybox
        command: ["sleep", "9999"]
---
apiVersion: v1
kind: Pod
metadata:
  name: spread-application-pod
  labels:
    app: spread-application
spec:
  containers:
  - name: application
    image: busybox
    command: ["sleep", "9999"]
```
