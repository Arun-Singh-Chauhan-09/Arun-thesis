apiVersion: v1
kind: Pod
metadata:
  name: ssd-pod
spec:
  containers:
  - name: ssd-container
    image: busybox
    command: ["sleep", "9999"]
  tolerations:
  - key: dedicated
    operator: Equal
    value: batch
    effect: NoSchedule
  nodeSelector:
    disktype: ssd

apiVersion: v1
kind: Pod
metadata:
  name: pod-with-pod-affinity
spec:
  affinity:
    nodeAffinity:
      requiredDuringSchedulingIgnoredDuringExecution:
        nodeSelectorTerms:
        - matchExpressions:
          - key: disktype
            operator: In
            values:
            - ssd
  containers:
  - name: container
    image: busybox
    command: ["sleep", "9999"]
  tolerations:
  - key: dedicated
    operator: Equal
    value: batch
    effect: NoSchedule
