```yaml
apiVersion: v1
kind: Secret
metadata:
  name: db-password
type: Opaque
data:
  DB_PASSWORD: <base!64 encoded password>
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: app
spec:
  replicas: 1
  selector:
    matchLabels:
      app: app
  template:
    metadata:
      labels:
        app: app
    spec:
      containers:
      - name: app
        image: busybox
        env:
        - name: DB_PASSWORD
          valueFrom:
            secretKeyRef:
              name: db-password
              key: DB_PASSWORD
      volumes: []
```
However, DB_PASSWORD value should be base64 encoded to store it as a Secret.

If you want to create the Secret from a literal string, you can use the following:

```yaml
apiVersion: v1
kind: Secret
metadata:
  name: db-password
type: Opaque
stringData:
  DB_PASSWORD: mysecretpassword
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: app
spec:
  replicas: 1
  selector:
    matchLabels:
      app: app
  template:
    metadata:
      labels:
        app: app
    spec:
      containers:
      - name: app
        image: busybox
        env:
        - name: DB_PASSWORD
          valueFrom:
            secretKeyRef:
              name: db-password
              key: DB_PASSWORD
      volumes: []
```
