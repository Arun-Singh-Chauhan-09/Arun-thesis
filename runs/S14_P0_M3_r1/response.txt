```yaml
apiVersion: v1
kind: Pod
metadata:
  name: app-with-sidecar
  labels:
    app: app-with-sidecar
spec:
  containers:
    - name: main-app
      image: busybox:1.36
      command: ["/bin/sh", "-c"]
      args:
        - >
          while true; do
            echo "$(date) - log entry from main container" >> /var/log/app/app.log;
            sleep 5;
          done
      volumeMounts:
        - name: shared-logs
          mountPath: /var/log/app

    - name: log-sidecar
      image: busybox:1.36
      command: ["/bin/sh", "-c"]
      args:
        - "tail -n+1 -F /var/log/app/app.log"
      volumeMounts:
        - name: shared-logs
          mountPath: /var/log/app
          readOnly: true

  volumes:
    - name: shared-logs
      emptyDir: {}
```
