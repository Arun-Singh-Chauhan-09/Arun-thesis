```yaml
apiVersion: batch/v1
kind: Job
metadata:
  name: db-migration
spec:
  backoffLimit: 2
  template:
    spec:
      containers:
      - name: db-migration
        image: migrate/migrate:v4.17.0
        command:
        - migrate
        - -path
        - /migrations
        - -database
        - $(DATABASE_URL)
        - up
        env:
        - name: DATABASE_URL
          value: "postgres://user:password@db-host:5432/dbname?sslmode=disable"
      restartPolicy: Never
```
