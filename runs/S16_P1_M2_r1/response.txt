```yaml
apiVersion: v1
kind: Service
metadata:
  name: database
spec:
  clusterIP: None
  selector:
    app: database
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: app
spec:
  selector:
    matchLabels:
      app: app
  template:
    metadata:
      labels:
        app: app
    spec:
      initContainers:
      - name: wait-for-db
        image: busybox:1.28
        command:
        - sh
        - -c
        - |
          until nslookup database; do
            echo "waiting for database service..."
            sleep 2
          done
      containers:
      - name: app
        image: nginx:1.14.2
        ports:
        - containerPort: 80
```
